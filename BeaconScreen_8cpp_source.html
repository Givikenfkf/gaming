53void BeaconScreen::tick() {
54 if (buttonsNotDrawn && beacon->getLevels() >= 0) {
55 buttonsNotDrawn = false;
56
57 int xo = (width - imageWidth) / 2;
58 int yo = (height - imageHeight) / 2;
59
60 for (int tier = 0; tier <= 2; ++tier) {
61 int effectCount = BeaconTileEntity::BEACON_EFFECTS_EFFECTS;
62 int actualCount = 0;
63 for (int e = 0; e < effectCount; ++e) {
64 if (BeaconTileEntity::BEACON_EFFECTS[tier][e] != nullptr) {
65 actualCount++;
66 } else {
67 break;
68 }
69 }
70
71 int totalWidth = actualCount * 22 + (actualCount - 1) * 2;
72 int startX = xo + 53 + (actualCount * 24 - totalWidth) / 2;
73
74 for (int e = 0; e < actualCount; ++e) {
75 MobEffect* effect = BeaconTileEntity::BEACON_EFFECTS[tier][e];
76 if (effect == nullptr) break;
77
78 int buttonId = (tier << 8) | effect->id;
79 BeaconPowerButton* button = new BeaconPowerButton(
80 this, buttonId, startX + e * 24, yo + 22 + tier * 25,
81 effect->id, tier);
82 buttons.push_back(button);
83
84 if (tier >= beacon->getLevels()) {
85 button->active = false;
86 } else if (effect->id == beacon->getPrimaryPower()) {
87 button->setSelected(true);
88 }
89 }
90 }
91
92 int tier = 3;
93 int effectCount = BeaconTileEntity::BEACON_EFFECTS_EFFECTS;
94 int actualCount = 0;
95 for (int e = 0; e < effectCount; ++e) {
96 if (BeaconTileEntity::BEACON_EFFECTS[tier][e] != nullptr) {
97 actualCount++;
98 } else {
99 break;
100 }
101 }
102
103 int totalWidth = (actualCount + 1) * 22 + actualCount * 2;
104 int startX = xo + 143 + ((actualCount + 1) * 24 - totalWidth) / 2;
105
106 for (int e = 0; e < actualCount; ++e) {
107 MobEffect* effect = BeaconTileEntity::BEACON_EFFECTS[tier][e];
108 if (effect == nullptr) break;
109
110 int buttonId = (tier << 8) | effect->id;
111 BeaconPowerButton* button = new BeaconPowerButton(
112 this, buttonId, startX + e * 24, yo + 47, effect->id, tier);
113 buttons.push_back(button);
114
115 if (tier >= beacon->getLevels()) {
116 button->active = false;
117 } else if (effect->id == beacon->getSecondaryPower()) {
118 button->setSelected(true);
119 }
120 }
121
122 if (beacon->getPrimaryPower() > 0) {
123 int buttonId = (tier << 8) | beacon->getPrimaryPower();
124 BeaconPowerButton* button =
125 new BeaconPowerButton(this, buttonId, startX + actualCount * 24,
126 yo + 47, beacon->getPrimaryPower(), tier);
127 buttons.push_back(button);
128
129 if (tier >= beacon->getLevels()) {
130 button->active = false;
131 } else if (beacon->getPrimaryPower() ==
132 beacon->getSecondaryPower()) {
133 button->setSelected(true);
134 }
135 }
136 }
137
138 beaconConfirmButton->active =
139 (beacon->getItem(0) != nullptr && beacon->getPrimaryPower() > 0);
140}

198void BeaconScreen::buttonClicked(Button* button) {
199 if (button->id == -2) {
200 minecraft->player->closeContainer();
201 } else if (button->id == -1) {
202 // 4jcraft: copied from IUIScene_BeaconMenu
203 ByteArrayOutputStream baos;
204 DataOutputStream dos(&baos);
205 dos.writeInt(beacon->getPrimaryPower());
206 dos.writeInt(beacon->getSecondaryPower());
207
208 minecraft->player->connection->send(
209 std::shared_ptr<CustomPayloadPacket>(new CustomPayloadPacket(
210 CustomPayloadPacket::SET_BEACON_PACKET, baos.toByteArray())));
211 minecraft->player->closeContainer();
212 } else if (dynamic_cast<BeaconPowerButton*>(button)) {
213 int effectId = button->id & 255;
214 int tier = button->id >> 8;
215
216 if (tier < 3) {
217 beacon->setPrimaryPower(effectId);
218 } else {
219 beacon->setSecondaryPower(effectId);
220 }
221
222 for (Button* btn : buttons) {
223 delete btn;
224 }
225 buttons.clear();
226 init();
227 tick();
228 }
