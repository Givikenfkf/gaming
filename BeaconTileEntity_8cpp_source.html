65void BeaconTileEntity::applyEffects() {
66 if (isActive && levels > 0 && !level->isClientSide && primaryPower > 0) {
67 double range = (levels * 10) + 10;
68 int baseAmp = 0;
69 if (levels >= 4 && primaryPower == secondaryPower) {
70 baseAmp = 1;
71 }
72
73 AABB bb = AABB(x, y, z, x + 1, y + 1, z + 1).grow(range, range, range);
74 bb.y1 = level->getMaxBuildHeight();
75 std::vector<std::shared_ptr<Entity> >* players =
76 level->getEntitiesOfClass(typeid(Player), &bb);
77 for (auto it = players->begin(); it != players->end(); ++it) {
78 std::shared_ptr<Player> player =
79 std::dynamic_pointer_cast<Player>(*it);
80 player->addEffect(new MobEffectInstance(
81 primaryPower, SharedConstants::TICKS_PER_SECOND * 9, baseAmp,
82 true));
83 }
84
85 if (levels >= 4 && primaryPower != secondaryPower &&
86 secondaryPower > 0) {
87 for (auto it = players->begin(); it != players->end(); ++it) {
88 std::shared_ptr<Player> player =
89 std::dynamic_pointer_cast<Player>(*it);
90 player->addEffect(new MobEffectInstance(
91 secondaryPower, SharedConstants::TICKS_PER_SECOND * 9, 0,
92 true));
93 }
94 }
95 delete players;
96 }
97}

99void BeaconTileEntity::updateShape() {
100 if (!level->canSeeSky(x, y + 1, z)) {
101 isActive = false;
102 levels = 0;
103 } else {
104 isActive = true;
105
106 levels = 0;
107 for (int step = 1; step <= 4; step++) {
108 int ly = y - step;
109 if (ly < 0) {
110 break;
111 }
112
113 bool isOk = true;
114 for (int lx = x - step; lx <= x + step && isOk; lx++) {
115 for (int lz = z - step; lz <= z + step; lz++) {
116 int tile = level->getTile(lx, ly, lz);
117 if (tile != Tile::emeraldBlock_Id &&
118 tile != Tile::goldBlock_Id &&
119 tile != Tile::diamondBlock_Id &&
120 tile != Tile::ironBlock_Id) {
121 isOk = false;
122 break;
123 }
124 }
125 }
126 if (isOk) {
127 levels = step;
128 } else {
129 break;
130 }
131 }
132 if (levels == 0) {
133 isActive = false;
134 }
135 }
136}

138float BeaconTileEntity::getAndUpdateClientSideScale() {
139 if (!isActive) {
140 return 0;
141 }
142
143 int renderDelta = (int)(level->getGameTime() - clientSideRenderTick);
144 clientSideRenderTick = level->getGameTime();
145 if (renderDelta > 1) {
146 clientSideRenderScale -= ((float)renderDelta / (float)SCALE_TIME);
147
148 if (clientSideRenderScale < 0) {
149 clientSideRenderScale = 0;
150 }
151 }
152 clientSideRenderScale += (1.0f / (float)SCALE_TIME);
153 if (clientSideRenderScale > 1) {
154 clientSideRenderScale = 1;
155 }
156 return clientSideRenderScale;
